function(check_closure target)
    if(NOT TARGET "${target}")
        message(FATAL_ERROR "fixture: cached package lost target ${target}")
    endif()
    string(MAKE_C_IDENTIFIER "${target}" key)
    get_property(visited GLOBAL PROPERTY "FIXTURE_CACHE_VISITED_${key}")
    if(visited)
        return()
    endif()
    set_property(GLOBAL PROPERTY "FIXTURE_CACHE_VISITED_${key}" TRUE)
    get_target_property(links "${target}" INTERFACE_LINK_LIBRARIES)
    foreach(link IN LISTS links)
        if(TARGET "${link}")
            check_closure("${link}")
        elseif(link MATCHES "::" OR IS_ABSOLUTE "${link}" OR link MATCHES "^\\$<")
            message(FATAL_ERROR "fixture: cached ${target} has unresolved link ${link}")
        endif()
    endforeach()
endfunction()
