cmake_minimum_required(VERSION 3.25)
if(NOT CASE MATCHES "^(runtime|scheduler|render|installed)$")
    message(FATAL_ERROR "fixture: unknown language CASE")
endif()
if(CASE STREQUAL "runtime")
    # A pure C consumer must not inspect or require a C++ compiler.
    set(CMAKE_CXX_COMPILER "${CMAKE_CURRENT_BINARY_DIR}/missing-cxx" CACHE FILEPATH "" FORCE)
endif()
project(tooling_language_fixture LANGUAGES C)
if(NOT IS_ABSOLUTE "${CHIBI_ROOT}" OR NOT EXISTS "${CHIBI_ROOT}/test/tools/languages.cmake")
    message(FATAL_ERROR "fixture: CHIBI_ROOT must identify the engine checkout")
endif()
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(CASE STREQUAL "scheduler" OR CASE STREQUAL "render")
    add_subdirectory(package)
elseif(CASE STREQUAL "installed")
    # Installed target discovery does not activate a language in its consumer.
    add_library(Chibi::render INTERFACE IMPORTED)
endif()

include("${CHIBI_ROOT}/test/tools/languages.cmake")
if(CASE STREQUAL "runtime")
    get_property(languages GLOBAL PROPERTY ENABLED_LANGUAGES)
    if("CXX" IN_LIST languages OR CMAKE_CXX_COMPILER_LOADED)
        message(FATAL_ERROR "fixture: pure C tooling enabled CXX")
    endif()
    add_library(fixture_consumer STATIC consumer.c)
elseif(CASE STREQUAL "scheduler")
    if(NOT CMAKE_CXX_COMPILER_LOADED OR NOT CMAKE_CXX_COMPILE_OBJECT)
        message(FATAL_ERROR "fixture: parent tooling lacks its CXX rules")
    endif()
    add_library(fixture_consumer STATIC consumer.c)
    target_link_libraries(fixture_consumer PRIVATE Chibi::task_scheduler)
else()
    # Reuse the allowed pass-through shim; generate its real C++ compile rule.
    # This fixture does not build the renderer or execute any produced code.
    add_library(fixture_consumer OBJECT "${CHIBI_ROOT}/test/draw_backend.cpp")
    target_link_libraries(fixture_consumer PRIVATE Chibi::render)
endif()
