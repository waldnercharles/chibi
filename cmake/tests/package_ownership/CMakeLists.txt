cmake_minimum_required(VERSION 3.25)
project(package_ownership C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_JOB_POOL_COMPILE console)
set(CMAKE_JOB_POOL_LINK console)
set(CHIBI_SDK_BUILD_NATIVE ON CACHE BOOL "")
set(CHIBI_SDK_BUILD_COMPILER ON CACHE BOOL "")
set(CHIBI_SDK_BUILD_DESKTOP OFF CACHE BOOL "")
add_library(folded_object OBJECT src/object.c)
add_library(folded_static STATIC src/folded.c)
add_library(owner STATIC src/owned.c $<TARGET_OBJECTS:folded_object> $<TARGET_OBJECTS:folded_static>)
target_precompile_headers(owner PRIVATE src/owned.h)
set_target_properties(owner PROPERTIES OUTPUT_NAME renamed ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/relocated")
add_library(debug_archive STATIC src/debug.c)
add_executable(auxiliary src/auxiliary.c)
if(CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin)$")
    add_library(chibi_build_host OBJECT src/debug.c)
    add_executable(chibi-build-host src/auxiliary.c)
    target_link_libraries(chibi-build-host PRIVATE chibi_build_host)
    add_library(chibi_compiler_host STATIC src/debug.c)
    add_executable(chibi-compiler-host src/auxiliary.c)
    target_link_libraries(chibi-compiler-host PRIVATE chibi_compiler_host)
    # A wrong executable owner must not satisfy the private-tool contract.
    target_link_libraries(auxiliary PRIVATE chibi_build_host)
    install(TARGETS chibi_compiler_host ARCHIVE DESTINATION lib)
    install(TARGETS chibi-build-host chibi-compiler-host RUNTIME DESTINATION libexec/chibi)
endif()
# Keep a .res-shaped compiler output with real host compilation and Ninja
# dependencies. This checks ownership, not native Windows resource compilation.
configure_file(src/resource.manifest resource.manifest COPYONLY)
set_source_files_properties(src/main.res PROPERTIES LANGUAGE C KEEP_EXTENSION ON
    OBJECT_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/resource.manifest")
add_executable(chibi-compile src/main.res)
target_link_libraries(chibi-compile PRIVATE owner)
install(TARGETS owner debug_archive ARCHIVE DESTINATION lib)
set_target_properties(chibi-compile PROPERTIES
    BUILD_RPATH "${CMAKE_BINARY_DIR}/build-only-provider"
    INSTALL_RPATH "$ORIGIN/../lib")
include("${CHIBI_ROOT}/sdk/products.cmake")
chibi_install_product(chibi-compile)
get_target_property(install_rpath chibi-compile INSTALL_RPATH)
if(NOT install_rpath STREQUAL "$ORIGIN/../lib")
    message(FATAL_ERROR "fixture: product install RPATH changed")
endif()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/owned.c" source_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/owned.h" header_hash)
file(WRITE "${CMAKE_BINARY_DIR}/chibi_engine_metadata.json"
    "{\"sdk_version\":\"0.1.0\",\"project_version\":1,\"artifact_formats\":{\"execution\":{\"version\":1,\"source\":\"src/owned.c\",\"header\":\"src/owned.h\",\"sha256\":\"${source_hash}\",\"header_sha256\":\"${header_hash}\"}}}\n")
include("${CHIBI_ROOT}/sdk/package_targets.cmake")
chibi_package_targets()
