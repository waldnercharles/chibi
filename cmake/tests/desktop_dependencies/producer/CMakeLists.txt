cmake_minimum_required(VERSION 3.24)
project(desktop_dependency_producer LANGUAGES NONE)
if(NOT IS_ABSOLUTE "${CHIBI_ROOT}" OR NOT EXISTS "${CHIBI_ROOT}/sdk/vendor_dependencies.cmake")
    message(FATAL_ERROR "fixture: CHIBI_ROOT must identify the engine checkout")
endif()
if(NOT CASE MATCHES "^(automatic|supplied|mismatch|opengl|unowned|archive)$")
    message(FATAL_ERROR "fixture: unknown CASE '${CASE}'")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../common.cmake")
set(CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../modules")
set(s2n_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../upstream")
set(FIXTURE_PROVIDER_ROOT "${CMAKE_CURRENT_BINARY_DIR}/provider")
fixture_inputs("${FIXTURE_PROVIDER_ROOT}")
add_library(fixture_sdk STATIC IMPORTED GLOBAL)
set_target_properties(fixture_sdk PROPERTIES
    IMPORTED_LOCATION "${FIXTURE_PROVIDER_ROOT}/lib/libfixture.a"
    INTERFACE_INCLUDE_DIRECTORIES "${FIXTURE_PROVIDER_ROOT}/include/sdk")

set(preserved_properties TYPE IMPORTED IMPORTED_GLOBAL SOURCE_DIR BINARY_DIR
    FIXTURE_ID IMPORTED_LOCATION INTERFACE_INCLUDE_DIRECTORIES
    INTERFACE_LINK_LIBRARIES INTERFACE_COMPILE_OPTIONS INTERFACE_LINK_OPTIONS
    INTERFACE_COMPILE_DEFINITIONS)
function(snapshot target)
    string(MAKE_C_IDENTIFIER "${target}" key)
    foreach(property IN LISTS preserved_properties)
        get_target_property(value "${target}" "${property}")
        set("before_${key}_${property}" "${value}" PARENT_SCOPE)
    endforeach()
endfunction()
function(check_preserved target)
    if(NOT TARGET "${target}")
        message(FATAL_ERROR "fixture: supplied target ${target} disappeared")
    endif()
    string(MAKE_C_IDENTIFIER "${target}" key)
    foreach(property IN LISTS preserved_properties)
        get_target_property(value "${target}" "${property}")
        if(NOT "${value}" STREQUAL "${before_${key}_${property}}")
            message(FATAL_ERROR "fixture: supplied ${target} ${property} was changed")
        endif()
    endforeach()
endfunction()

if(CASE STREQUAL "supplied")
    add_library(Threads::Threads INTERFACE IMPORTED)
    set_target_properties(Threads::Threads PROPERTIES
        FIXTURE_ID "caller-threads" INTERFACE_LINK_LIBRARIES "-pthread"
        INTERFACE_COMPILE_OPTIONS "-pthread" INTERFACE_COMPILE_DEFINITIONS "CALLER_THREADS=1")
    add_library(ZLIB::ZLIB STATIC IMPORTED)
    set_target_properties(ZLIB::ZLIB PROPERTIES
        FIXTURE_ID "caller-zlib" IMPORTED_LOCATION "${FIXTURE_PROVIDER_ROOT}/lib/libz.a"
        INTERFACE_INCLUDE_DIRECTORIES "${FIXTURE_PROVIDER_ROOT}/include/zlib"
        INTERFACE_COMPILE_DEFINITIONS "CALLER_ZLIB=1")
    add_library(AWS::crypto STATIC IMPORTED)
    set_target_properties(AWS::crypto PROPERTIES
        FIXTURE_ID "caller-crypto" IMPORTED_LOCATION "${FIXTURE_PROVIDER_ROOT}/lib/libcrypto.a"
        INTERFACE_INCLUDE_DIRECTORIES "${FIXTURE_PROVIDER_ROOT}/include/crypto"
        INTERFACE_LINK_LIBRARIES "Threads::Threads;ZLIB::ZLIB;dl"
        INTERFACE_COMPILE_DEFINITIONS "CALLER_CRYPTO=1")
    foreach(target AWS::crypto Threads::Threads ZLIB::ZLIB)
        fixture_property("${target}" IMPORTED_GLOBAL "FALSE")
        snapshot("${target}")
    endforeach()
endif()

if(CASE MATCHES "^(opengl|unowned)$")
    find_package(OpenGL REQUIRED)
    set(links "${OPENGL_gl_LIBRARY};${OPENGL_opengl_LIBRARY};${OPENGL_glx_LIBRARY}")
    if(CASE STREQUAL "unowned")
        list(APPEND links "${FIXTURE_PROVIDER_ROOT}/unowned/libGL.so")
    endif()
    set_property(TARGET fixture_sdk PROPERTY INTERFACE_LINK_LIBRARIES "${links}")
elseif(CASE STREQUAL "archive")
    # A C++ library from the third-party archive, whose licences the archive
    # recorded, and one spelled License beside its headers.
    set(CHIBI_THIRD_PARTY_DIR "${FIXTURE_PROVIDER_ROOT}/archive")
    file(WRITE "${CHIBI_THIRD_PARTY_DIR}/share/chibi/licenses/vendor/fixture_sdk/LICENSE" "archive\n")
    file(WRITE "${FIXTURE_PROVIDER_ROOT}/include/License.txt" "beside\n")
    set_property(TARGET fixture_sdk PROPERTY IMPORTED_LINK_INTERFACE_LANGUAGES CXX)
else()
    add_subdirectory(crypto)
endif()

include("${CHIBI_ROOT}/sdk/vendor_dependencies.cmake")
chibi_vendor_dependency(fixture_sdk exported)
if(CASE STREQUAL "unowned")
    message(FATAL_ERROR "fixture: same-basename unowned OpenGL path was accepted")
endif()
file(READ "${CHIBI_VENDOR_DEPENDENCY_CONFIG}" config)
if(CASE MATCHES "^(automatic|supplied)$")
    # The pre-fix exporter silently leaves AWS::crypto unresolved in this scope.
    if(NOT config MATCHES "add_library\\(Chibi::vendor_AWS__crypto ")
        message(FATAL_ERROR "fixture: missing owned crypto target after child-scoped discovery")
    endif()
    if(NOT config MATCHES "add_library\\(Chibi::vendor_ZLIB__ZLIB ")
        message(FATAL_ERROR "fixture: missing owned zlib target in crypto closure")
    endif()
endif()
if(CASE STREQUAL "automatic")
    get_property(discovered GLOBAL PROPERTY FIXTURE_OPENSSL_DISCOVERED)
    if(NOT discovered)
        message(FATAL_ERROR "fixture: static OpenSSL metadata was not exercised")
    endif()
endif()
if(CASE STREQUAL "supplied")
    foreach(target AWS::crypto Threads::Threads ZLIB::ZLIB)
        check_preserved("${target}")
    endforeach()
    get_property(discovered GLOBAL PROPERTY FIXTURE_OPENSSL_DISCOVERED)
    if(discovered)
        message(FATAL_ERROR "fixture: caller provider triggered OpenSSL fallback")
    endif()
endif()
foreach(forbidden "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}" "${CHIBI_ROOT}")
    string(FIND "${config}" "${forbidden}" found)
    if(NOT found EQUAL -1)
        message(FATAL_ERROR "fixture: dependency config retains producer path ${forbidden}")
    endif()
endforeach()
install(FILES "${CHIBI_VENDOR_DEPENDENCY_CONFIG}" DESTINATION lib/cmake/Chibi)
