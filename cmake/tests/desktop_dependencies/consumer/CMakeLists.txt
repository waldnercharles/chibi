# Inspect the entire closure explicitly: interface-only generation misses unknown
# namespaced libraries, stale archive paths, and missing installed headers.
function(inspect target)
    if(NOT TARGET "${target}")
        message(FATAL_ERROR "fixture: missing installed dependency ${target}")
    endif()
    string(MAKE_C_IDENTIFIER "${target}" key)
    get_property(visited GLOBAL PROPERTY "FIXTURE_VISITED_${key}")
    if(visited)
        return()
    endif()
    set_property(GLOBAL PROPERTY "FIXTURE_VISITED_${key}" TRUE)
    get_target_property(type "${target}" TYPE)
    if(target MATCHES "^Chibi::vendor_")
        if(NOT type STREQUAL "INTERFACE_LIBRARY")
            get_target_property(archive "${target}" IMPORTED_LOCATION)
            string(REGEX REPLACE "^Chibi::vendor_" "" owner "${target}")
            set(expected "${PREFIX}/lib/chibi-vendor/libchibi_vendor_${owner}.a")
            if(NOT archive STREQUAL expected OR NOT EXISTS "${archive}")
                message(FATAL_ERROR "fixture: ${target} archive is not owned by relocated SDK: ${archive}")
            endif()
            file(READ "${archive}" contents)
            if(NOT contents STREQUAL "!<arch>\n")
                message(FATAL_ERROR "fixture: ${target} archive was not copied intact")
            endif()
        endif()
        get_target_property(includes "${target}" INTERFACE_INCLUDE_DIRECTORIES)
        if(NOT includes)
            message(FATAL_ERROR "fixture: ${target} lost its public headers")
        endif()
        if(target STREQUAL "Chibi::vendor_fixture_sdk")
            set(header fixture.h)
            set(expected_header "#define FIXTURE_SDK 1\n")
        elseif(target STREQUAL "Chibi::vendor_AWS__crypto")
            set(header openssl/crypto.h)
            set(expected_header "#define FIXTURE_CRYPTO 1\n")
        elseif(target STREQUAL "Chibi::vendor_ZLIB__ZLIB")
            set(header zlib.h)
            set(expected_header "#define FIXTURE_ZLIB 1\n")
        else()
            message(FATAL_ERROR "fixture: unexpected SDK owner ${target}")
        endif()
        foreach(path IN LISTS includes)
            string(FIND "${path}" "${PREFIX}/include/chibi/vendor/" owned)
            if(NOT owned EQUAL 0 OR NOT IS_DIRECTORY "${path}")
                message(FATAL_ERROR "fixture: ${target} include path is not relocated: ${path}")
            endif()
            if(NOT EXISTS "${path}/${header}")
                message(FATAL_ERROR "fixture: ${target} public header was not installed")
            endif()
            file(READ "${path}/${header}" contents)
            if(NOT contents STREQUAL expected_header)
                message(FATAL_ERROR "fixture: ${target} public header identity changed")
            endif()
        endforeach()
    endif()
    get_target_property(links "${target}" INTERFACE_LINK_LIBRARIES)
    foreach(link IN LISTS links)
        if(link MATCHES "^\\$<LINK_ONLY:(.*)>$")
            set(link "${CMAKE_MATCH_1}")
        endif()
        if(TARGET "${link}")
            inspect("${link}")
        elseif(link MATCHES "::" OR IS_ABSOLUTE "${link}" OR link MATCHES "^\\$<")
            message(FATAL_ERROR "fixture: ${target} has unresolved or nonportable link ${link}")
        endif()
    endforeach()
endfunction()

    foreach(pair "LICENSE|archive" "License.txt|beside")
        string(REPLACE "|" ";" pair "${pair}")
        list(GET pair 0 name)
        list(GET pair 1 expected)
        if(NOT EXISTS "${licenses}/${name}")
            message(FATAL_ERROR "fixture: the SDK did not install fixture_sdk's ${name}")
        endif()
        file(READ "${licenses}/${name}" contents)
        if(NOT contents STREQUAL "${expected}\n")
            message(FATAL_ERROR "fixture: fixture_sdk's ${name} is not the ${expected} licence")
        endif()
    endforeach()
